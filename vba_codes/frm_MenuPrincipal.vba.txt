Private Sub btn_NavClientes_Click()
    On Error GoTo TrataErro
    ' Abre o formulário de clientes e garante que ele ganhe o foco
    DoCmd.OpenForm "frm_Clientes", acNormal
    
Exit_Sub:
    Exit Sub
TrataErro:
    MsgBox "Erro ao abrir a tela de Clientes: " & Err.Description, vbCritical, "ConfecTech"
    Resume Exit_Sub
End Sub
